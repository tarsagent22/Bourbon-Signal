const first = fixture(0), second = fixture(1);
const options = { now: '2026-10-23T01:11:00.000Z', allowedBottleIds: ['synthetic-eagle'] };
const later = structuredClone(second.queueProof);

const assemble = proofs => assembleSoakEvidence([first.captured, second.captured], {}, { ...options, queueProof: proofs });

  assert.ok(mapProjectedCandidates(sourceId, [first.candidate.dedupeKey], [first.observation], [second.candidate],
    { runId: first.trace.run_id, generation: first.trace.generation }).errors.some(error => error.includes('historical remapping')));
});

  assert.ok(mapProjectedCandidates(sourceId, [first.candidate.dedupeKey], [first.observation], [first.candidate],
    { runId: first.trace.run_id, generation: first.trace.generation + 1 }).errors.length);

  const result = assemble([first.queueProof, later]);
  assert.equal(result.evidence.sources[0].runs.length, 2);
  assert.ok(result.evidence.sources[0].runs.every(run => run.projectedCandidateCount === 1 && run.candidateCount === 0));
